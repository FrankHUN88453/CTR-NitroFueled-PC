cmake_minimum_required(VERSION 3.25)
project(ctr_pc CXX RC)

# the game executable (icon + version info in res/app.rc)
set(APP CrashTeamRacingNitroFueled)

if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "ctr_pc needs clang-cl (sysv_abi calling convention for guest calls)")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(${APP}
    src/main.cpp
    src/log.cpp
    src/nid.cpp
    src/module.cpp
    src/linker.cpp
    src/thread.cpp
    src/hle.cpp
    src/crash.cpp
    src/memory.cpp
    src/patches.cpp
    src/config.cpp
    src/prompts.cpp
    src/mouse_ui.cpp
    src/settings_menu.cpp
    src/hle/kernel_misc.cpp
    src/hle/kernel_thread.cpp
    src/hle/kernel_file.cpp
    src/hle/system.cpp
    src/hle/kernel_event.cpp
    src/hle/gnm.cpp
    src/hle/videoout.cpp
    src/hle/audio.cpp
    src/hle/pad.cpp
    src/hle/savedata.cpp
    src/hle/playgo.cpp
    src/gfx/probe.cpp
    src/gfx/formats.cpp
    src/gfx/renderer.cpp
    src/gfx/texture.cpp
    src/gfx/glyphs.cpp
    res/app.rc
)
target_include_directories(${APP} PRIVATE src)
target_compile_definitions(${APP} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(${APP} PRIVATE /EHsc /Z7 -Wno-unused-function -Wno-unused-variable)
target_link_options(${APP} PRIVATE /DEBUG)
target_link_libraries(${APP} PRIVATE user32 gdi32 dbghelp xinput d3d11 dxgi d3dcompiler gdiplus ole32)

# the embedded shader blob is regenerated by build.bat; rebuild the resources when it changes
set_source_files_properties(res/app.rc PROPERTIES OBJECT_DEPENDS
    "${CMAKE_SOURCE_DIR}/res/shaders.bin;${CMAKE_SOURCE_DIR}/res/glyphs_xbox.rgba")
