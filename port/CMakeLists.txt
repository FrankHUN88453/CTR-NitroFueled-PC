cmake_minimum_required(VERSION 3.25)
if(WIN32)
    project(ctr_pc CXX RC)
else()
    project(ctr_pc CXX)
endif()

# the game executable (Windows: icon + version info in res/app.rc)
set(APP CrashTeamRacingNitroFueled)

if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "ctr_pc needs clang / clang-cl (sysv_abi calling convention for guest calls)")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(${APP}
    src/main.cpp
    src/log.cpp
    src/nid.cpp
    src/module.cpp
    src/linker.cpp
    src/thread.cpp
    src/hle.cpp
    src/crash.cpp
    src/memory.cpp
    src/patches.cpp
    src/config.cpp
    src/prompts.cpp
    src/mouse_ui.cpp
    src/settings_menu.cpp
    src/first_start.cpp
    src/selftest.cpp
    src/hle/kernel_misc.cpp
    src/hle/kernel_thread.cpp
    src/hle/kernel_file.cpp
    src/hle/system.cpp
    src/hle/kernel_event.cpp
    src/hle/gnm.cpp
    src/hle/videoout.cpp
    src/hle/audio.cpp
    src/hle/pad.cpp
    src/hle/savedata.cpp
    src/hle/playgo.cpp
    src/gfx/probe.cpp
    src/gfx/formats.cpp
    src/gfx/renderer.cpp
    src/gfx/texture.cpp
    src/gfx/glyphs.cpp
)
target_include_directories(${APP} PRIVATE src)

# CTR_EMBED_DATA (default): the translated shaders and the Xbox glyphs go into the exe (local
# builds, regenerated by build.bat). OFF for public releases: then the exe holds nothing of the
# games: on the first start the setup embedded instead (res/setup.bin, tools/pack_setup.py) makes
# shaders.bin / glyphs_xbox.rgba from the player's dump (first_start.cpp). Linux builds never
# embed them: the AppImage carries the setup as files.
option(CTR_EMBED_DATA "embed the translated shaders and button glyphs" ON)

if(WIN32)
    target_sources(${APP} PRIVATE res/app.rc)
    target_compile_definitions(${APP} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${APP} PRIVATE /EHsc /Z7 -Wno-unused-function -Wno-unused-variable)
    target_link_options(${APP} PRIVATE /DEBUG)
    target_link_libraries(${APP} PRIVATE user32 gdi32 dbghelp xinput d3d11 dxgi d3dcompiler gdiplus ole32)
    if(CTR_EMBED_DATA)
        set_source_files_properties(res/app.rc PROPERTIES OBJECT_DEPENDS
            "${CMAKE_SOURCE_DIR}/res/shaders.bin;${CMAKE_SOURCE_DIR}/res/glyphs_xbox.rgba")
    else()
        set_source_files_properties(res/app.rc PROPERTIES COMPILE_DEFINITIONS CTR_NO_EMBED
            OBJECT_DEPENDS "${CMAKE_SOURCE_DIR}/res/setup.bin")
        target_link_options(${APP} PRIVATE "/PDBALTPATH:%_PDB%")  # no local build path in the exe
    endif()
else()
    # Linux: Win32 shim (POSIX + SDL2) and D3D11 through DXVK native. DXVK_INCLUDE_WINDOWS /
    # DXVK_INCLUDE_DIRECTX: its header folders (native/windows, native/directx), DXVK_LIB_DIR:
    # libdxvk_d3d11.so / libdxvk_dxgi.so.
    set(DXVK_INCLUDE_WINDOWS "" CACHE PATH "DXVK native: include/native/windows")
    set(DXVK_INCLUDE_DIRECTX "" CACHE PATH "DXVK native: include/native/directx")
    set(DXVK_LIB_DIR "" CACHE PATH "DXVK native: folder with libdxvk_d3d11.so")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    target_sources(${APP} PRIVATE
        src/platform/linux/win32_shim.cpp
        src/platform/linux/memory_watch.cpp
        src/platform/linux/sdl_shim.cpp
        src/platform/linux/crash_linux.cpp
        src/platform/linux/videoout_sdl.cpp
        src/platform/linux/first_start_linux.cpp
    )
    # our windows.h comes first and include_next's DXVK's
    target_include_directories(${APP} BEFORE PRIVATE src/platform/linux/include)
    target_include_directories(${APP} PRIVATE ${DXVK_INCLUDE_WINDOWS} ${DXVK_INCLUDE_DIRECTX})
    target_compile_options(${APP} PRIVATE -Wno-unused-function -Wno-unused-variable -Wno-ignored-attributes
                           -Wno-format -Wno-multichar -Wno-nonportable-include-path -fno-strict-aliasing)
    target_link_directories(${APP} PRIVATE ${DXVK_LIB_DIR})
    target_link_libraries(${APP} PRIVATE dxvk_d3d11 dxvk_dxgi PkgConfig::SDL2 pthread dl)
    # the AppImage keeps its libraries in usr/lib next to usr/bin
    set_target_properties(${APP} PROPERTIES BUILD_RPATH "\$ORIGIN/../lib" INSTALL_RPATH "\$ORIGIN/../lib")
endif()
